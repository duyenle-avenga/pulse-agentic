> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Set up the Pulse Agentic Next.js app, a mock data layer that derives AC states from raw PR evidence, and the Portfolio and Project overview screens.

**Architecture:** Async Server Components call a `PulseRepository` (`src/lib/data`). The mock implementation combines raw demo facts (`src/lib/mock`) with pure derivation functions (`src/lib/progress`) and returns plain view models. No API routes. Only the burn‑up chart is a client component.

**Tech Stack:** Next.js (latest, App Router, TypeScript strict, `src/`), Tailwind CSS v4, shadcn/ui, Recharts via shadcn `chart`, Vitest + Testing Library (jsdom), Playwright (Chromium).

**Spec:** `docs/superpowers/specs/2026-10-16-poc-foundation-design.md`

**Small changes compared with the spec**, decided while planning:
- The repository has a third method, `getProjectHeader(id)`, so the project layout doesn't compute the whole overview.
- The 404 page lives at `src/app/not-found.tsx`, not under `projects/[projectId]`. Next.js routes a `notFound()` thrown in a layout to the *parent* segment's boundary, so a not-found file next to the layout would never catch it.
- Activity events have no link target yet. None of the screens they would link to exist.
- Project C tasks have no `externalKey`. The scope doc doesn't say which CTOACT ticket matches which task, so we don't invent a mapping.
- Dark mode follows the OS setting (`prefers-color-scheme`), with no toggle.

````markdown
# Architecture

How the Pulse Agentic POC code is organised. For *what* the product does, read `README.md`.

## Folder structure

```
src/
  app/                          Next.js App Router (routes only, thin)
    layout.tsx                  top bar with the demo clock
    not-found.tsx               404 page
    page.tsx                    Portfolio
    projects/[projectId]/
      layout.tsx                project header + sidebar
      page.tsx                  Project overview
  components/
    ui/                         shadcn/ui primitives (generated; don't edit by hand)
    portfolio/                  Portfolio screen components
    project/                    Project screens' components
    labels.ts                   display labels and colours for domain enums
  lib/
    domain/                     domain types and small helpers (no I/O)
    progress/                   pure derivation: AC states, rollups, burn-up, activity
    data/                       repository interface + implementations (the only door to data)
    mock/                       demo data for projects A, B and C, and the demo clock
    format.ts                   date and duration formatting
e2e/                            Playwright smoke tests
docs/                           design docs, specs and plans
```

## Data flow

```
page.tsx (async Server Component)
  → getRepository()                 src/lib/data/index.ts
  → MockRepository                  src/lib/data/mock-repository.ts
      → raw facts                   src/lib/mock/*
      → deriveAcStates, rollup,     src/lib/progress/*
        burnup, recentActivity
  → plain view models (props)
  → components
```

Rules:
- **No API routes yet.** Pages call the repository directly. Later write flows (user testing, approvals) will use Server Actions over the same repository.
- **The UI never imports `lib/mock`.** It goes through `lib/data`, so a Postgres implementation can replace the mock without UI changes.
- **Mocks hold raw facts only** (scope, tasks, parsed PRs, human test results, SCRs, signals). AC states and all numbers are derived in `lib/progress` (README §3.3).
- **`lib/progress` is pure:** no I/O, no real clock. "Now" is passed in (`DEMO_NOW` in the POC).
- Server Components by default. Only components that need the browser (charts) are client components.

## Tests

- `npm test`: Vitest unit and component tests, next to the code (`*.test.ts(x)`).
- `npm run test:e2e`: Playwright smoke tests against a production build.
````

```ts
import type { AcceptanceCriterion, Project, ScopeVersion } from './types'

export function currentScope(project: Project): ScopeVersion {
  const version = project.scopeVersions.at(-1)
  if (!version) throw new Error(`Project ${project.id} has no scope version`)
  return version
}

export function acsOf(version: ScopeVersion): AcceptanceCriterion[] {
  return version.features.flatMap((f) => f.stories.flatMap((s) => s.acs))
}
```

```ts
import type {
  AcceptanceCriterion,
  Feature,
  IsoDateTime,
  PrSections,
  PullRequest,
  Task,
  TestPlanItem,
  TestType,
  UserStory,
} from '@/lib/domain/types'

const pad = (n: number) => String(n).padStart(2, '0')

/** A 2026 timestamp in Europe/Berlin summer time (all demo dates are before 25 Oct). */
export const at = (month: number, day: number, hour = 9, minute = 0): IsoDateTime =>
  `2026-${pad(month)}-${pad(day)}T${pad(hour)}:${pad(minute)}:00+02:00`

export const ac = (id: string, text: string, humanTest = false): AcceptanceCriterion => ({ id, text, humanTest })
export const story = (id: string, text: string, acs: AcceptanceCriterion[]): UserStory => ({ id, text, acs })
export const feature = (id: string, title: string, stories: UserStory[]): Feature => ({ id, title, stories })

type TaskInput = Omit<Task, 'projectId' | 'intent' | 'dependsOn'> & Partial<Pick<Task, 'intent' | 'dependsOn'>>

export const task = (projectId: string, input: TaskInput): Task => ({ intent: '', dependsOn: [], ...input, projectId })

export const ticked = (acId: string, testType: TestType, testFile?: string): TestPlanItem => ({
  acId, text: acId, mark: 'ticked', testType, testFile,
})
export const unticked = (acId: string): TestPlanItem => ({ acId, text: acId, mark: 'unticked' })
export const failedItem = (acId: string, testType: TestType, testFile?: string): TestPlanItem => ({
  acId, text: acId, mark: 'failed', testType, testFile,
})

type PrInput = Pick<PullRequest, 'number' | 'title' | 'state' | 'openedAt' | 'ci'> &
  Partial<Omit<PullRequest, 'projectId' | 'sections'>> & { sections?: Partial<PrSections> }

/** Deterministic fake commit SHA so mocks stay stable. */
const fakeSha = (n: number) => ((n * 2654435761) >>> 0).toString(16).padStart(8, '0').slice(0, 7)

export function pr(projectId: string, input: PrInput): PullRequest {
  const testPlan = input.testPlan ?? []
  const scopeChangeNotes = input.scopeChangeNotes ?? []
  return {
    taskIds: [],
    headSha: fakeSha(input.number),
    ...input,
    testPlan,
    scopeChangeNotes,
    sections: {
      summary: true,
      migration: false,
      testPlan: testPlan.length > 0,
      notes: scopeChangeNotes.length > 0,
      ...input.sections,
    },
    projectId,
  }
}
```

```ts
import type { IsoDateTime } from '@/lib/domain/types'

const TIME_ZONE = 'Europe/Berlin'
// Our own month names: en-GB ICU data prints "Sept" for September.
const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']

const partsFormatter = new Intl.DateTimeFormat('en-GB', {
  timeZone: TIME_ZONE,
  year: 'numeric',
  month: 'numeric',
  day: 'numeric',
  hour: '2-digit',
  minute: '2-digit',
  hourCycle: 'h23',
})

function berlinParts(at: IsoDateTime) {
  const parts = partsFormatter.formatToParts(new Date(at))
  const get = (type: Intl.DateTimeFormatPartTypes) => parts.find((p) => p.type === type)?.value ?? ''
  return { year: get('year'), month: Number(get('month')), day: Number(get('day')), time: `${get('hour')}:${get('minute')}` }
}

/** "22 Sep" */
export function formatDate(at: IsoDateTime): string {
  const p = berlinParts(at)
  return `${p.day} ${MONTHS[p.month - 1]}`
}

/** "16 Oct 2026, 09:00" */
export function formatDateTime(at: IsoDateTime): string {
  const p = berlinParts(at)
  return `${p.day} ${MONTHS[p.month - 1]} ${p.year}, ${p.time}`
}

/** "25 min", "3 h", "4 d" */
export function formatDuration(since: IsoDateTime, now: IsoDateTime): string {
  const minutes = Math.max(0, Math.floor((Date.parse(now) - Date.parse(since)) / 60_000))
  if (minutes < 60) return `${minutes} min`
  const hours = Math.floor(minutes / 60)
  if (hours < 48) return `${hours} h`
  return `${Math.floor(hours / 24)} d`
}

/** "3 h ago", "just now" */
export function formatRelative(at: IsoDateTime, now: IsoDateTime): string {
  const duration = formatDuration(at, now)
  return duration === '0 min' ? 'just now' : `${duration} ago`
}
```

```ts
import { acsOf, currentScope } from '@/lib/domain/scope'
import type { AcState, HumanTestResult, IsoDateTime, Project, PullRequest, Task } from '@/lib/domain/types'

export interface AcTransition {
  state: AcState
  at: IsoDateTime
}

export interface AcStatus {
  acId: string
  humanTest: boolean
  state: AcState
  /** Every state change, oldest first. Empty means not_started. */
  history: AcTransition[]
}

export interface EvidenceInput {
  project: Project
  tasks: Task[]
  prs: PullRequest[]
  humanTests: HumanTestResult[]
}

type AcEvent =
  | { kind: 'opened'; at: IsoDateTime; order: 0 }
  | { kind: 'merged'; at: IsoDateTime; order: 1; pr: PullRequest }
  | { kind: 'tested'; at: IsoDateTime; order: 2; result: HumanTestResult }

const NOT_YET_DELIVERED: readonly AcState[] = ['not_started', 'in_progress']

/** Applies README §3.3 to raw evidence. See the spec §4.1 for the exact rules. */
export function deriveAcStates({ project, tasks, prs, humanTests }: EvidenceInput): AcStatus[] {
  const acIdsByTask = new Map(tasks.map((t) => [t.id, new Set(t.acIds)]))
  const touches = (pr: PullRequest, acId: string) =>
    pr.taskIds.some((id) => acIdsByTask.get(id)?.has(acId)) || pr.testPlan.some((item) => item.acId === acId)

  return acsOf(currentScope(project)).map((ac) => {
    const events: AcEvent[] = []
    for (const pr of prs) {
      if (pr.state === 'closed' || !touches(pr, ac.id)) continue
      events.push({ kind: 'opened', at: pr.openedAt, order: 0 })
      if (pr.state === 'merged' && pr.mergedAt) events.push({ kind: 'merged', at: pr.mergedAt, order: 1, pr })
    }
    if (ac.humanTest) {
      for (const result of humanTests) {
        if (result.acId === ac.id) events.push({ kind: 'tested', at: result.testedAt, order: 2, result })
      }
    }
    events.sort((a, b) => Date.parse(a.at) - Date.parse(b.at) || a.order - b.order)

    let state: AcState = 'not_started'
    const history: AcTransition[] = []
    for (const event of events) {
      const next = ac.humanTest ? nextHumanState(state, event) : nextRegularState(state, event, ac.id)
      if (next !== state) {
        state = next
        history.push({ state, at: event.at })
      }
    }
    return { acId: ac.id, humanTest: ac.humanTest, state, history }
  })
}

function nextRegularState(state: AcState, event: AcEvent, acId: string): AcState {
  if (event.kind === 'opened') return state === 'not_started' ? 'in_progress' : state
  if (event.kind !== 'merged') return state

  const item = event.pr.testPlan.find((i) => i.acId === acId)
  const ticked = item?.mark === 'ticked'
  if (item?.mark === 'failed' || (ticked && event.pr.ci === 'failing')) {
    return state === 'verified' || state === 'regressed' ? 'regressed' : 'failing'
  }
  if (ticked && (event.pr.ci === 'passing' || event.pr.ci === 'none')) return 'verified'
  // Unticked, missing from the test plan, or CI still pending: merged but not proven.
  return NOT_YET_DELIVERED.includes(state) ? 'implemented' : state
}

function nextHumanState(state: AcState, event: AcEvent): AcState {
  if (event.kind === 'opened') return state === 'not_started' ? 'in_progress' : state
  if (event.kind === 'merged') return NOT_YET_DELIVERED.includes(state) ? 'pending_user_testing' : state
  return event.result.result === 'pass' ? 'verified' : 'failing'
}
```

```ts
import { AC_STATES, type AcState, type IsoDateTime } from '@/lib/domain/types'
import type { AcStatus } from './deriveAcStates'

/** "Implemented or better" (spec §4.2). Failing and regressed count as risk instead. */
export const DELIVERED_STATES: ReadonlySet<AcState> = new Set<AcState>(['implemented', 'pending_user_testing', 'verified'])

export interface Rollup {
  total: number
  counts: Record<AcState, number>
  completionPct: number
  deliveryPct: number
  unverifiedPct: number
  atRisk: number
  pendingUserTesting: number
  oldestPendingSince: IsoDateTime | null
}

export function rollup(statuses: AcStatus[]): Rollup {
  const counts = Object.fromEntries(AC_STATES.map((s) => [s, 0])) as Record<AcState, number>
  for (const status of statuses) counts[status.state] += 1

  const total = statuses.length
  const pct = (n: number) => (total === 0 ? 0 : Math.round((n / total) * 100))
  const completionPct = pct(counts.verified)
  const deliveryPct = pct(statuses.filter((s) => DELIVERED_STATES.has(s.state)).length)

  const pendingSince = statuses
    .filter((s) => s.state === 'pending_user_testing')
    .map((s) => s.history.at(-1)?.at)
    .filter((at): at is IsoDateTime => at !== undefined)
    .sort((a, b) => Date.parse(a) - Date.parse(b))

  return {
    total,
    counts,
    completionPct,
    deliveryPct,
    unverifiedPct: deliveryPct - completionPct,
    atRisk: counts.failing + counts.regressed,
    pendingUserTesting: counts.pending_user_testing,
    oldestPendingSince: pendingSince[0] ?? null,
  }
}
```

```ts
import type { AcState, IsoDateTime, Project } from '@/lib/domain/types'
import type { AcStatus } from './deriveAcStates'
import { DELIVERED_STATES } from './rollup'

export interface BurnupPoint {
  /** YYYY-MM-DD */
  date: string
  scope: number
  delivered: number
  verified: number
}

const DAY_MS = 86_400_000
// Days end at midnight Europe/Berlin summer time, matching the demo data.
const TZ_OFFSET = '+02:00'

export function stateAt(status: AcStatus, cutoffMs: number): AcState {
  let state: AcState = 'not_started'
  for (const t of status.history) {
    if (Date.parse(t.at) > cutoffMs) break
    state = t.state
  }
  return state
}

const nextDay = (day: string) => new Date(Date.parse(`${day}T00:00:00Z`) + DAY_MS).toISOString().slice(0, 10)

export function burnup(project: Project, statuses: AcStatus[], now: IsoDateTime): BurnupPoint[] {
  const versions = project.scopeVersions
  if (versions.length === 0) return []
  const nowMs = Date.parse(now)
  const lastDay = now.slice(0, 10)

  const points: BurnupPoint[] = []
  for (let day = versions[0].lockedAt.slice(0, 10); day <= lastDay; day = nextDay(day)) {
    const cutoff = Math.min(Date.parse(`${day}T23:59:59${TZ_OFFSET}`), nowMs)
    const active = versions.filter((v) => Date.parse(v.lockedAt) <= cutoff).at(-1) ?? versions[0]
    const states = statuses.map((s) => stateAt(s, cutoff))
```

```ts
import type { IsoDateTime, ProjectData } from '@/lib/domain/types'

export type ActivityKind = 'scope_locked' | 'pr_opened' | 'pr_merged' | 'human_test' | 'scr_raised' | 'scr_decided'

export interface ActivityEvent {
  at: IsoDateTime
  kind: ActivityKind
  label: string
}

export function recentActivity(data: ProjectData, limit = 10): ActivityEvent[] {
  const events: ActivityEvent[] = []

  for (const v of data.project.scopeVersions) {
    events.push({ at: v.lockedAt, kind: 'scope_locked', label: `Scope ${v.label} locked by ${v.approvedBy}` })
  }
  for (const pr of data.prs) {
    events.push({ at: pr.openedAt, kind: 'pr_opened', label: `PR #${pr.number} opened: ${pr.title}` })
    if (pr.mergedAt) events.push({ at: pr.mergedAt, kind: 'pr_merged', label: `PR #${pr.number} merged: ${pr.title}` })
  }
  for (const t of data.humanTests) {
    const outcome = t.result === 'pass' ? 'passed' : 'failed'
    events.push({ at: t.testedAt, kind: 'human_test', label: `${t.acId} human test ${outcome} (${t.tester}, ${t.environment})` })
  }
  for (const scr of data.scrs) {
    events.push({ at: scr.raisedAt, kind: 'scr_raised', label: `${scr.id} raised: ${scr.title}` })
    if (scr.status !== 'pending' && scr.decidedAt) {
      events.push({ at: scr.decidedAt, kind: 'scr_decided', label: `${scr.id} ${scr.status} by ${scr.decidedBy ?? 'unknown'}: ${scr.title}` })
    }
  }

  return events.sort((a, b) => Date.parse(b.at) - Date.parse(a.at)).slice(0, limit)
}
```

```ts
import { currentScope } from '@/lib/domain/scope'
import type { IsoDateTime, ProjectData, Severity } from '@/lib/domain/types'
import { recentActivity } from '@/lib/progress/activity'
import { burnup } from '@/lib/progress/burnup'
import { deriveAcStates } from '@/lib/progress/deriveAcStates'
import { rollup } from '@/lib/progress/rollup'
import type { ProjectHeaderData, PulseRepository } from './repository'

const SEVERITY_RANK: Record<Severity, number> = { high: 0, medium: 1, low: 2 }

function headerOf({ project, scrs }: ProjectData): ProjectHeaderData {
  const scope = currentScope(project)
  return {
    id: project.id,
    name: project.name,
    description: project.description,
    kind: project.kind,
    board: project.board,
    boardKey: project.boardKey,
    repoUrl: project.repoUrl,
    scopeLabel: scope.label,
    lockedAt: scope.lockedAt,
    approvedBy: scope.approvedBy,
    pendingApprovals: scrs.filter((s) => s.status === 'pending').length,
  }
}

export function createMockRepository(projects: ProjectData[], now: IsoDateTime): PulseRepository {
  const find = (id: string) => projects.find((d) => d.project.id === id)

  return {
    async getPortfolio() {
      return projects.map((data) => {
        const header = headerOf(data)
        const r = rollup(deriveAcStates(data))
        return {
          id: header.id,
          name: header.name,
          kind: header.kind,
          board: header.board,
          boardKey: header.boardKey,
          scopeLabel: header.scopeLabel,
          completionPct: r.completionPct,
          deliveryPct: r.deliveryPct,
          verified: r.counts.verified,
          total: r.total,
          openSignals: data.signals.length,
          highSignals: data.signals.filter((s) => s.severity === 'high').length,
          pendingApprovals: header.pendingApprovals,
          pendingUserTesting: r.pendingUserTesting,
          lastActivityAt: recentActivity(data, 1)[0]?.at ?? null,
        }
      })
    },

    async getProjectHeader(projectId) {
      const data = find(projectId)
      return data ? headerOf(data) : null
    },

    async getProjectOverview(projectId) {
      const data = find(projectId)
      if (!data) return null
      const statuses = deriveAcStates(data)
      const topSignals = [...data.signals]
        .sort((a, b) => SEVERITY_RANK[a.severity] - SEVERITY_RANK[b.severity] || Date.parse(a.raisedAt) - Date.parse(b.raisedAt))
        .slice(0, 5)
      return {
        header: headerOf(data),
        rollup: rollup(statuses),
        burnup: burnup(data.project, statuses, now),
        topSignals,
        activity: recentActivity(data),
      }
    },
  }
}
```
